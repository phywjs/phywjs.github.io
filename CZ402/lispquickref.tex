\sect{Arithmetic}
\line{\hskip 3mm {\tt +}, {\tt -}, {\tt *}, {\tt /}, 
{\tt expt} (power) \hfill
\quad e.g. {\tt (+ (* a (expt x 2)) (* 2 x) (/ -3 5))} 
\quad $a x^2 + 2x - 3/5$\hskip 3mm}

\sect{Data Objects}
\settabs 5 \columns
\+\fct{123}{integer} &
\fct{3/7}{ratio} &
\fct{1.3}{floating-point} &
\fct{\#C(5 -3)}{$5 - 3i$} &
\fct{"text"}{string} &\cr
\settabs 2 \columns
\+\fct{(a b c)}{list with elements {\tt a}, {\tt b}, {\tt c}.} &
\fct{(a\ .\ b)}{dotted pair, same as {\tt (cons 'a 'b)}} & \cr
\+\fct{\#(a b c)}{a vector of length 3} &
\fct{(aref a $i$ $j$)}{array $a_{i,j}$} & \cr
\+\fct{(make-array '(3 3))}{create $3\times 3$ array} &
\fct{(defstruct $foo$ $a$ $b$)}{define structure} &\cr

\sect{List Manipulation}
\settabs 2 \columns
\+\fct{(car '(a b c)) {\rm or} (first '(a b c))}{return {\tt a}} &
\fct{(cdr '(a b c)) {\rm or} (rest '(a b c))}{give {\tt (b c)}} & \cr
\+\fct{(cons 'a '(b))}{return {\tt (a b)}} &
\fct{(list 'a 'b)}{return {\tt (a b)}} & \cr
\+\fct{(second '(a b c)) {\rm or} (nth 1 '(a b c))}{return {\tt b}} &
\fct{(third '(a b c)) {\rm or} (nth 2 '(a b c))}{return {\tt c}} & \cr
\+\fct{(last '(a b c))}{return {\tt (c)}} &
\fct{(nthcdr 2 '(a b c d))}{remove from front {\tt (c d)}} & \cr
\+\fct{(butlast '(a b c d) 2)}{remove from end {\tt (a b)}} &
\fct{(reverse '(a b c))}{return {\tt (c b a)}} & \cr
\+\fct{(list-length $list$)}{number of elements} &
\fct{(append '(a) '(b c))}{return {\tt (a b c)}} & \cr
\+\fct{(push 'a $list$)}{add {\tt a} to $list$} &
\fct{(pop $list$)}{remove top element} & \cr

\sect{Defining Variables and Functions}
\settabs 2 \columns
\+\fct{(setq x 5)}{bind a value to symbol $x$} &
\fct{(setf x 5)}{a more general assignment} & \cr
\+\fct{(defun f (x) $body$)}{function definition} & 
\fct{(let ((x 1) (y 2)) $body$)}{bind values locally} &\cr
\+\fct{(defun f (x \&optional (y 0)) $body$)}{{\tt y} is optional} & 
\fct{(defmacro f (x) $body$)}{macro definition}&\cr
\+\fct{(defun f (x \&rest y) $body$)}{any number of arguments} & &\cr

\sect{Predicates}
\settabs 3 \columns
\+\fct{nil {\rm or} ()}{for false} &
\fct{t {\rm or anything else}}{for true} & 
\fct{(null $x$)}{is $x$ empty list?} &\cr
\+\fct{(atom $a$)}{is $a$ an atom?} &
\fct{(listp $x$)}{is $x$ a list?} &
\fct{(numberp $x$)}{is $x$ a number?} &\cr
\+\fct{(integerp $x$)}{is $x$ an integer?} &
\fct{(rationalp $x$)}{is $x$ a rational?} &
\fct{(floatp $x$)}{is $x$ a float?} &\cr

\sect{Conditional and Iterations}
\settabs 1 \columns
\+\fct{(if $test$ $then$ $else$)}{if $test$ is true, evaluate $then$,
                            (optionally) otherwise, evaluate $else$}&\cr
\+\fct{(when $test$ $form$)}{evaluate $form$ only if $test$ is {\tt t}}&\cr
\+\fct{(unless $test$ $form$)}{evaluate $form$ only if $test$ is 
                                                      {\tt nil}}&\cr
\+\fct{(cond  ($test_1$ $form_1$) ($test_2$ $form_2$) ...)}{if $test_1$
is {\tt t}, evaluate $form_1$, else if $test_2$ is {\tt t}, 
evalute $form_2$, ...}&\cr
\+\fct{(loop $form$)}{infinite loop {\tt return} for termination}&\cr
\+\fct{(do ((x 1 (+ x 1)) ($test$) $forms$)}{do-loop}&\cr
\+\fct{(dolist (x '(a b c) ) (print x))}{{\tt x} runs over elements 
                                           of the list}&\cr
\+\fct{(progn $form_1$ $form_2$ $form_3$)}{for grouping, where a single
form is required}&\cr

\sect{Logical Operators}
\settabs 4 \columns
\+\fct{(eq $a$ $b$)}{same address} &
\fct{(eql $a$ $b$)}{same value} &
\fct{equal}{same structure} &
\fct{equalp}{weak equal} & \cr
\+\fct{<}{less than} &
\fct{<=}{less or equal} &
\fct{(or $a$ $b$)}{logical or} & 
\fct{(and $a$ $b$)}{logical and} & \cr
\+\fct{>}{greater than} &
\fct{>=}{greater or equal} &
\fct{(not $a$)}{negate} &
\fct{(oddp $a$)}{is $a$ odd?} & \cr
\+\fct{=}{number equal} &
\fct{/=}{number differ} &
\fct{(zerop $a$)}{$a=0$?} &
\fct{(evenp $a$)}{is $a$ even?} & \cr
