Bitwise Operator, examples

    #include <stdio.h>


    main()

    {

    short int w1, w2;

    w1 = 12;

    w2 = -35;

    printf("w1 = %d\n", w1);

    printf("w2 = %d\n", w2);

    printf("~w1 = %d\n", ~w1);

    printf("~w2 = %d\n", ~w2);

    printf("w1 & w2 = %d\n", w1 & w2);

    printf("w1 & ~w2 = %d\n", w1 & ~w2);

    printf("~w1 & ~w2 = %d\n", ~w1 & ~w2);

    printf("w1 | w2 = %d\n", w1 | w2);

    printf("~(w1 | w2) = %d\n", ~(w1 | w2));

    printf("w1 ^ w2 = %d\n", w1 ^ w2);

    }

    This output is

    w1 = 12

    w2 = -35

    ~w1 = -13

    ~w2 = 34

    w1 & w2 = 12

    w1 & ~w2 = 0

    ~w1 & ~w2 = 34

    w1 | w2 = -35

    ~(w1 | w2) = 34

    w1 ^ w2 = -47
