    void update_lhs(void)


    {

    char *next = buffer, *last;

    char op[10], numStr[20];

    int num;

    int temp = lhs;

    last = next + strlen(buffer);

    while (next < last) {

    if ( sscanf(next, "%s %s", op, numStr) < 2 ||

    strlen(op) != 1 ||

    bad_op(op[0]) )

    return;

    if (Stop == op[0] ) {

    printf(" *** End of Calculation ***\n\n");

    exit(EXIT_SUCCESS);

    }

    num = atoi(numStr);

    switch(op[0]) {

    case '+': temp += num; break;

    case '-': temp -= num; break;

    case '*': temp *= num; break;

    case '/': temp /= num; break;

    case '%': temp %= num; break;

    }

    next += strlen(op) + strlen(numStr) + 2;

    }

    lhs = temp;

    }
