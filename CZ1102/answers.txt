double fabs(double x)
{
   return (x>=0) ? x : -x;
}

#include <float.h>
double square_root(double x)
{
   double g = 1.0;
   while( fabs((g*g-x)/x) > DBL_EPSILON) {
     g = 0.5*(x/g + g);
   }
   return g;
}
