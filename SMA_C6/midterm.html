3.
function [mean, var, max, index] = stat(A);
% STAT takes a matrix and returns its mean, variance, maximum and index.
% We run over the elements and compute required quantities.
mean = 0.0;
square = 0.0;
max = -inf;
index = [-1, -1];
[m, n] = size(A);
for i = 1:m
    for j = 1:n
        v = A(i,j);
        mean = mean + v;
        square = square + v*v;
        if (A(i,j) > max) 
            max = v;
            index = [i j];
        end
    end
end;
mean = mean/(n*m);
var = square/(n*m) - mean*mean;
